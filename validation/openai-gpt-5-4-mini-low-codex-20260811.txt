    function parseCsv(text) {
      const rows = text.trim().split(/\r?\n/);

    function mean(values) {
      if (!values.length) return 0;
      return values.reduce((sum, v) => sum + v, 0) / values.length;
    }

    function stddev(values) {
      if (values.length < 2) return 0;
      const m = mean(values);
      const variance = values.reduce((sum, v) => sum + (v - m) ** 2, 0) / (values.length - 1);
      return Math.sqrt(Math.max(0, variance));

    function linearRegression(xs, ys) {
      const n = xs.length;
      const xMean = mean(xs);
      const yMean = mean(ys);
      let numerator = 0;
      let denominator = 0;
      for (let i = 0; i < n; i++) {
        numerator += (xs[i] - xMean) * (ys[i] - yMean);
        denominator += (xs[i] - xMean) ** 2;
      }
      const slope = denominator === 0 ? 0 : numerator / denominator;
      const intercept = yMean - slope * xMean;
      const fitted = xs.map(x => intercept + slope * x);
      const residuals = ys.map((y, i) => y - fitted[i]);
      const ssRes = residuals.reduce((sum, v) => sum + v * v, 0);
      const ssTot = ys.reduce((sum, y) => sum + (y - yMean) ** 2, 0);
      const r2 = ssTot === 0 ? 1 : 1 - (ssRes / ssTot);
      return { slope, intercept, residuals, r2 };
    }

    function addBusinessDays(startTime, days) {

        const day = date.getUTCDay();
        if (day !== 0 && day !== 6) added++;
      }
      return date.getTime();
    }

    function buildMovingAverage(series, window) {
      const out = [];
      for (let i = 0; i < series.length; i++) {

    function fitModel(series, horizon, lookback, model) {
      const window = series.slice(-Math.max(5, Math.min(lookback, series.length)));
      const closes = window.map(d => d.close);
      const xs = window.map((_, i) => i);
      const logPrices = closes.map(v => Math.log(v));
      const reg = linearRegression(xs, logPrices);
      const logResidualVol = Math.max(stddev(reg.residuals), 0.0015);
      const returns = [];
      for (let i = 1; i < closes.length; i++) {
        returns.push(Math.log(closes[i] / closes[i - 1]));
      }
      const returnVol = Math.max(stddev(returns), 0.006);
      const recentMeanReturn = mean(returns.slice(-Math.min(20, returns.length)));
      const longMean = mean(closes.slice(-Math.min(200, closes.length)));
      const lastClose = closes[closes.length - 1];
      const actualTrend = Math.exp(reg.slope * 252) - 1;
      const trendStrength = Math.min(1, Math.max(0, Math.abs(reg.slope) * 3500));
      const forecast = [];
      const checkpoints = [];
      const z = 1.96;
      const bandVol = Math.max(logResidualVol, returnVol);

      for (let step = 1; step <= horizon; step++) {
        const idx = window.length - 1 + step;
        let predicted;
        if (model === 'trend') {
          predicted = Math.exp(reg.intercept + reg.slope * idx);
        } else if (model === 'momentum') {
          predicted = lastClose * Math.exp(recentMeanReturn * step);
        } else {
          const tau = Math.max(10, Math.min(45, lookback / 12));
          predicted = longMean + (lastClose - longMean) * Math.exp(-step / tau);
        }

        const horizonVol = bandVol * Math.sqrt(step);
        const lower = predicted * Math.exp(-z * horizonVol);
        const upper = predicted * Math.exp(z * horizonVol);
        const time = addBusinessDays(window[window.length - 1].time, step);
        forecast.push({
          step,
          time,
          predicted,
          lower,
          upper,
          kind: 'forecast',
        });
      }

      for (const point of forecast.slice(0, 10)) {
        checkpoints.push(point);
      }

      const latest = closes[closes.length - 1];
      const lastYear = series.findIndex(d => d.time >= addBusinessDays(series[series.length - 1].time, -252));
      const yearSlice = lastYear >= 0 ? series.slice(Math.max(0, lastYear)) : series.slice(0);
      const yearReturn = yearSlice.length > 1 ? (latest / yearSlice[0].close) - 1 : 0;

      return {
        model,
        lookback: window.length,
        closeSeries: series,
        actualWindow: window,

    }

        `;
      }).join('');
    }

    function buildPaths(points, yScale, xScale) {
      return points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${xScale(i).toFixed(2)} ${yScale(p).toFixed(2)}`).join(' ');
    }

    function renderChart(model) {
      const svg = els.chart;
      svg.innerHTML = '';

      const actual = model.actualWindow.map((d, idx) => ({
        kind: 'actual',
        date: d.date,
        time: d.time,
        value: d.close,
        index: idx,
      }));
      const ma50 = buildMovingAverage(model.actualWindow, Math.min(50, model.actualWindow.length)).map((value, idx) => ({
        kind: 'ma',
        time: model.actualWindow[idx].time,
        value,
        index: idx,
      }));
      const forecast = model.forecast.map((d, idx) => ({
        kind: 'forecast',
        time: d.time,
        value: d.predicted,
        lower: d.lower,
        upper: d.upper,
        index: actual.length + idx,
      }));

      const allPoints = actual.concat(forecast);
      const allValues = [];
      for (const p of actual) allValues.push(p.value);
      for (const p of forecast) {
        allValues.push(p.lower, p.upper);
      }
      for (const p of ma50) allValues.push(p.value);

      const width = 1100;
      const height = 500;
      const margin = { top: 18, right: 26, bottom: 42, left: 68 };
      const plotWidth = width - margin.left - margin.right;
      const plotHeight = height - margin.top - margin.bottom;
      const minY = Math.min(...allValues);
      const maxY = Math.max(...allValues);
      const pad = (maxY - minY) * 0.06 || 1;
      const yMin = minY - pad;
      const yMax = maxY + pad;
      const xCount = Math.max(1, allPoints.length - 1);

      const xScale = index => margin.left + (index / xCount) * plotWidth;
      const yScale = value => margin.top + (1 - ((value - yMin) / (yMax - yMin))) * plotHeight;

      function add(el) {
        svg.appendChild(el);
        return el;
      }

      svg.setAttribute('viewBox', `0 0 ${width} ${height}`);
      svg.setAttribute('preserveAspectRatio', 'none');

      const gridCount = 5;
      for (let i = 0; i <= gridCount; i++) {
        const y = margin.top + (i / gridCount) * plotHeight;
        const val = yMax - ((y / height) * (yMax - yMin) / (plotHeight / height));
        const line = createSvgEl('line');
        line.setAttribute('x1', margin.left);
        line.setAttribute('x2', width - margin.right);
        line.setAttribute('y1', y);
        line.setAttribute('y2', y);
        line.setAttribute('class', 'grid-line');
        add(line);

        const label = createSvgEl('text');
        label.setAttribute('x', margin.left - 10);
        label.setAttribute('y', y + 4);
        label.setAttribute('text-anchor', 'end');
        label.setAttribute('class', 'axis-label');
        label.textContent = fmt.price.format(val);
        add(label);
      }

      const xTicks = 6;
      for (let i = 0; i <= xTicks; i++) {
        const idx = Math.round((i / xTicks) * xCount);
        const point = allPoints[idx];
        if (!point) continue;
        const x = xScale(idx);
        const line = createSvgEl('line');
        line.setAttribute('x1', x);
        line.setAttribute('x2', x);
        line.setAttribute('y1', margin.top);
        line.setAttribute('y2', height - margin.bottom);
        line.setAttribute('class', 'grid-line');
        add(line);

        const label = createSvgEl('text');
        label.setAttribute('x', x);
        label.setAttribute('y', height - 14);
        label.setAttribute('text-anchor', i === 0 ? 'start' : i === xTicks ? 'end' : 'middle');
        label.setAttribute('class', 'axis-label');
        label.textContent = businessDayLabel(point.time, i);
        add(label);
      }

      const axisBottom = createSvgEl('line');
      axisBottom.setAttribute('x1', margin.left);
      axisBottom.setAttribute('x2', width - margin.right);
      axisBottom.setAttribute('y1', height - margin.bottom);
      axisBottom.setAttribute('y2', height - margin.bottom);
      axisBottom.setAttribute('class', 'axis-line');
      add(axisBottom);

      const axisLeft = createSvgEl('line');
      axisLeft.setAttribute('x1', margin.left);
      axisLeft.setAttribute('x2', margin.left);
      axisLeft.setAttribute('y1', margin.top);
      axisLeft.setAttribute('y2', height - margin.bottom);
      axisLeft.setAttribute('class', 'axis-line');
      add(axisLeft);

      const bandPath = (() => {
        const upper = forecast.map(p => ({ value: p.upper }));
        const lower = forecast.map(p => ({ value: p.lower })).reverse();
        const upperPoints = actual.map(p => ({ value: p.value })).concat(upper);
        const lowerPoints = lower.concat(actual.slice().reverse().map(p => ({ value: p.value })));
        const upperLine = upperPoints.map((p, i) => `${i === 0 ? 'M' : 'L'} ${xScale(i).toFixed(2)} ${yScale(p.value).toFixed(2)}`).join(' ');
        const lowerLine = lowerPoints.map((p, i) => `${i === 0 ? 'M' : 'L'} ${xScale(upperPoints.length - 1 - i).toFixed(2)} ${yScale(p.value).toFixed(2)}`).join(' ');
        return `${upperLine} ${lowerLine} Z`;
      })();

      const band = createSvgEl('path');
      band.setAttribute('d', bandPath);
      band.setAttribute('class', 'band-path');
      add(band);

      const actualLine = createSvgEl('path');
      actualLine.setAttribute('d', buildPaths(actual.map(p => p.value), yScale, xScale));
      actualLine.setAttribute('class', 'actual-line');
      add(actualLine);

      const maLine = createSvgEl('path');
      maLine.setAttribute('d', buildPaths(ma50.map(p => p.value), yScale, index => xScale(index)));
      maLine.setAttribute('class', 'ma-line');
      add(maLine);

      const forecastLine = createSvgEl('path');
      forecastLine.setAttribute('d', buildPaths(forecast.map(p => p.value), yScale, index => xScale(actual.length - 1 + index)));
      forecastLine.setAttribute('class', 'forecast-line');
      add(forecastLine);

      const hoverLine = createSvgEl('line');
      hoverLine.setAttribute('class', 'hover-line');
      hoverLine.setAttribute('y1', margin.top);
      hoverLine.setAttribute('y2', height - margin.bottom);
      hoverLine.setAttribute('visibility', 'hidden');
      add(hoverLine);

      const hoverDot = createSvgEl('circle');
      hoverDot.setAttribute('r', 5.5);
      hoverDot.setAttribute('class', 'hover-dot');
      hoverDot.setAttribute('stroke', COLORS.accent);
      hoverDot.setAttribute('visibility', 'hidden');
      add(hoverDot);

      const overlay = createSvgEl('rect');
      overlay.setAttribute('x', margin.left);
      overlay.setAttribute('y', margin.top);
      overlay.setAttribute('width', plotWidth);
      overlay.setAttribute('height', plotHeight);
      overlay.setAttribute('fill', 'transparent');
      overlay.style.cursor = 'crosshair';
      add(overlay);

      const points = allPoints.map((p, idx) => ({
        ...p,
        index: idx,
        x: xScale(idx),
        y: yScale(p.value),
      }));

      function showTooltip(evt, point) {
        if (!point) {
          els.tooltip.style.opacity = 0;
          hoverLine.setAttribute('visibility', 'hidden');
          hoverDot.setAttribute('visibility', 'hidden');
          return;
        }
        const left = Math.min(evt.clientX - els.chartWrap.getBoundingClientRect().left + 18, els.chartWrap.clientWidth - 250);
        const top = Math.max(8, evt.clientY - els.chartWrap.getBoundingClientRect().top - 84);
        els.tooltip.style.left = `${left}px`;
        els.tooltip.style.top = `${top}px`;
        els.tooltip.style.opacity = 1;
        els.tooltip.innerHTML = `
          <div class="date">${formatDatePretty(point.time)}</div>
          <div class="value">${fmt.price.format(point.value)}</div>
          <div class="kind">${point.kind === 'forecast' ? 'Forecast' : point.kind === 'ma' ? '50-day MA' : 'Actual close'}</div>
        `;
        hoverLine.setAttribute('visibility', 'visible');
        hoverLine.setAttribute('x1', point.x);
        hoverLine.setAttribute('x2', point.x);
        hoverDot.setAttribute('visibility', 'visible');
        hoverDot.setAttribute('cx', point.x);
        hoverDot.setAttribute('cy', point.y);
      }

      function nearestPoint(x) {
        const idx = Math.max(0, Math.min(points.length - 1, Math.round((x - margin.left) / plotWidth * xCount)));
        return points[idx];
      }

      overlay.addEventListener('mousemove', evt => {
        const rect = svg.getBoundingClientRect();
        const x = evt.clientX - rect.left;
        showTooltip(evt, nearestPoint(x));
      });
      overlay.addEventListener('mouseleave', () => {
        els.tooltip.style.opacity = 0;
        hoverLine.setAttribute('visibility', 'hidden');
        hoverDot.setAttribute('visibility', 'hidden');
      });
    }

    function renderApp() {
      if (!allSeries.length) return;
      const historyDays = Number(els.historyRange.value);
      const modelName = els.modelSelect.value;
      const lookback = Number(els.lookback.value);
      const horizon = Number(els.horizon.value);

      els.lookbackValue.textContent = fmt.int.format(lookback);
      els.horizonValue.textContent = fmt.int.format(horizon);

      const history = chooseHistory(allSeries, historyDays);
      const actualWindow = history.length > 10 ? history : allSeries.slice();
      currentModel = fitModel(actualWindow, horizon, lookback, modelName);
      renderMetrics(currentModel);
      updateModelCopy(currentModel);
      buildForecastTable(currentModel);
      renderChart(currentModel);

      const sourceText = allSeries.length > actualWindow.length ? 'Live FRED data loaded' : 'Fallback sample loaded';
      setStatus(`${sourceText} · ${formatDatePretty(currentModel.lastDate)}`, false);
    }

    async function loadData() {
      setStatus('Loading live FRED series...', true);
      try {
        const response = await fetch(FRED_URL, { cache: 'no-store' });
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const text = await response.text();
        const parsed = parseCsv(text);
        if (!parsed.length) throw new Error('Empty series');
        allSeries = parsed;
        setStatus(`Live data loaded · ${formatDatePretty(parsed[parsed.length - 1].time)}`, false);
      } catch (err) {
        console.warn('FRED fetch failed, using fallback data:', err);
        allSeries = parseCsv(FALLBACK_CSV);
        setStatus(`Fallback sample loaded · ${formatDatePretty(allSeries[allSeries.length - 1].time)}`, false);
      }
      renderApp();
    }

    function bindEvents() {
      for (const el of [els.historyRange, els.modelSelect, els.lookback, els.horizon]) {
        el.addEventListener('input', () => {
          if (el === els.modelSelect) {
            const name = el.value;
            if (name === 'trend') {
              els.lookback.value = 252;
            } else if (name === 'momentum') {
              els.lookback.value = 120;
            } else if (name === 'meanreversion') {
              els.lookback.value = 252;
            }
          }
          renderApp();
        });
      }

      window.addEventListener('resize', () => {
        if (currentModel) renderChart(currentModel);
      });
    }

    bindEvents();
    loadData();
  </script>
